###Comandos para crear las imagenes ya que necesitan ser compiladas previamente en la etapa builder con:

# Construir imagen del servidor gRPC
docker build -f grpc_server/Dockerfile -t grpc-server-image .

# Construir imagen de la API REST
docker build -f api/Dockerfile -t rest-api-image .

### Correr los contenedores por separado con compose no hace falta el networks

#crear una red compartida
docker network create grpc-network


# Primero el servidor gRPC
docker run -d --name grpc-server --network grpc-network grpc-server-image

# Luego la API REST (que depende del anterior)
docker run -d --name rest-api -p 8081:8081 --network grpc-network rest-api-image
